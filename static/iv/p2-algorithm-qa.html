时间复杂度与空间复杂度

1. 什么是时间复杂度和空间复杂度？

答案：
时间复杂度：描述算法执行时间随输入规模增长的变化趋势，用大 O 符号表示（如 O(n)、O(log n)、O(n²)）。空间复杂度：描述算法运行所需额外内存的增长趋势。常见复杂度从优到劣：O(1) < O(log n) < O(n) < O(n log n) < O(n²) < O(2ⁿ)。面试中分析复杂度时关注最坏情况（Big O），忽略常数和低阶项。

2. 如何快速判断算法的时间复杂度？

答案：
经验规则：① 单层循环：O(n)；② 嵌套循环：O(n²)（或更高）；③ 每次缩小一半（二分）：O(log n)；④ 递归树展开：O(分支数^深度)；⑤ 排序算法：比较排序下界 O(n log n)。Map/Set 的增删查：O(1)均摊；数组按索引访问 O(1)、查找 O(n)；二叉搜索树均衡时 O(log n)。

数组与字符串

3. 两数之和（LeetCode 1）

题目： 给定整数数组和目标值，返回两数之和等于目标值的下标。
答案：

function twoSum(nums, target) {
  const map = new Map(); // value -> index
  for (let i = 0; i < nums.length; i++) {
    const complement = target - nums[i];
    if (map.has(complement)) return [map.get(complement), i];
    map.set(nums[i], i);
  }
}
// 时间 O(n)，空间 O(n)。暴力两层循环为 O(n²)。

4. 合并两个有序数组（LeetCode 88）

题目： 将 nums2 合并入 nums1，nums1 末尾有足够空间。
答案：

function merge(nums1, m, nums2, n) {
  let i = m - 1, j = n - 1, k = m + n - 1;
  while (i >= 0 && j >= 0) {
    nums1[k--] = nums1[i] > nums2[j] ? nums1[i--] : nums2[j--];
  }
  while (j >= 0) nums1[k--] = nums2[j--];
}
// 从后往前填充，避免覆盖，时间 O(m+n)，空间 O(1)。

5. 找到数组中的最大子数组和（Kadane 算法）

答案：

function maxSubArray(nums) {
  let maxSum = nums[0], currentSum = nums[0];
  for (let i = 1; i < nums.length; i++) {
    currentSum = Math.max(nums[i], currentSum + nums[i]);
    maxSum = Math.max(maxSum, currentSum);
  }
  return maxSum;
}
// 动态规划：dp[i] = max(nums[i], dp[i-1] + nums[i])，时间 O(n)，空间 O(1)。

6. 滑动窗口：无重复字符的最长子串（LeetCode 3）

答案：

function lengthOfLongestSubstring(s) {
  const set = new Set();
  let left = 0, maxLen = 0;
  for (let right = 0; right < s.length; right++) {
    while (set.has(s[right])) set.delete(s[left++]);
    set.add(s[right]);
    maxLen = Math.max(maxLen, right - left + 1);
  }
  return maxLen;
}
// 滑动窗口，时间 O(n)，空间 O(min(n, 字符集大小))。

7. 前缀和：子数组和等于 k（LeetCode 560）

答案：

function subarraySum(nums, k) {
  const map = new Map([[0, 1]]); // 前缀和 -> 出现次数
  let sum = 0, count = 0;
  for (const num of nums) {
    sum += num;
    count += map.get(sum - k) ?? 0;
    map.set(sum, (map.get(sum) ?? 0) + 1);
  }
  return count;
}
// 前缀和 + 哈希表，时间 O(n)，空间 O(n)。

链表

8. 反转链表（LeetCode 206）

答案：

function reverseList(head) {
  let prev = null, curr = head;
  while (curr) {
    const next = curr.next;
    curr.next = prev;
    prev = curr;
    curr = next;
  }
  return prev;
}
// 迭代：时间 O(n)，空间 O(1)。递归版：时间 O(n)，空间 O(n)（调用栈）。

9. 检测链表是否有环（LeetCode 141）

答案：

function hasCycle(head) {
  let slow = head, fast = head;
  while (fast && fast.next) {
    slow = slow.next;
    fast = fast.next.next;
    if (slow === fast) return true;
  }
  return false;
}
// Floyd 龟兔算法：快慢指针，时间 O(n)，空间 O(1)。

10. 找链表的中间节点

答案：

function middleNode(head) {
  let slow = head, fast = head;
  while (fast && fast.next) {
    slow = slow.next;
    fast = fast.next.next;
  }
  return slow; // 偶数节点时返回后半段第一个节点
}
// 快慢指针，快指针到达末尾时慢指针恰好在中间。

11. 合并两个有序链表（LeetCode 21）

答案：

function mergeTwoLists(l1, l2) {
  const dummy = { next: null };
  let curr = dummy;
  while (l1 && l2) {
    if (l1.val <= l2.val) { curr.next = l1; l1 = l1.next; }
    else { curr.next = l2; l2 = l2.next; }
    curr = curr.next;
  }
  curr.next = l1 || l2;
  return dummy.next;
}
// 哨兵节点简化边界，时间 O(m+n)，空间 O(1)。

树

12. 二叉树的三种遍历（前中后序）

答案：

// 中序遍历（左-根-右），递归版
function inorder(root, result = []) {
  if (!root) return result;
  inorder(root.left, result);
  result.push(root.val);
  inorder(root.right, result);
  return result;
}

// 中序遍历，迭代版（栈）
function inorderIterative(root) {
  const result = [], stack = [];
  let curr = root;
  while (curr || stack.length) {
    while (curr) { stack.push(curr); curr = curr.left; }
    curr = stack.pop();
    result.push(curr.val);
    curr = curr.right;
  }
  return result;
}

13. 二叉树的层序遍历（BFS，LeetCode 102）

答案：

function levelOrder(root) {
  if (!root) return [];
  const result = [], queue = [root];
  while (queue.length) {
    const levelSize = queue.length;
    const level = [];
    for (let i = 0; i < levelSize; i++) {
      const node = queue.shift();
      level.push(node.val);
      if (node.left) queue.push(node.left);
      if (node.right) queue.push(node.right);
    }
    result.push(level);
  }
  return result;
}
// BFS，时间 O(n)，空间 O(n)（队列最大为最宽一层节点数）。

14. 二叉树的最大深度（LeetCode 104）

答案：

function maxDepth(root) {
  if (!root) return 0;
  return 1 + Math.max(maxDepth(root.left), maxDepth(root.right));
}
// DFS 递归，时间 O(n)，空间 O(h)（h 为树高，最差 O(n) 退化为链）。

15. 验证二叉搜索树（LeetCode 98）

答案：

function isValidBST(root, min = -Infinity, max = Infinity) {
  if (!root) return true;
  if (root.val <= min || root.val >= max) return false;
  return isValidBST(root.left, min, root.val)
      && isValidBST(root.right, root.val, max);
}
// 传递有效值范围，时间 O(n)，空间 O(h)。

16. 前端场景：树形数据扁平化 / 构建树

答案：

// 列表 → 树（常见于后端返回扁平数据构建菜单树）
function buildTree(nodes) {
  const map = new Map(nodes.map(n => [n.id, { ...n, children: [] }]));
  const roots = [];
  for (const node of map.values()) {
    node.parentId ? map.get(node.parentId)?.children.push(node) : roots.push(node);
  }
  return roots;
}

// 树 → 列表（DFS 扁平化）
function flatTree(nodes, result = []) {
  for (const node of nodes) {
    result.push(node);
    if (node.children?.length) flatTree(node.children, result);
  }
  return result;
}

动态规划

17. 爬楼梯（LeetCode 70）

题目： n 阶楼梯，每次爬 1 或 2 阶，有多少种方法？
答案：

function climbStairs(n) {
  if (n <= 2) return n;
  let a = 1, b = 2;
  for (let i = 3; i <= n; i++) {
    [a, b] = [b, a + b];
  }
  return b;
}
// Fibonacci 数列，空间优化到 O(1)，本质 dp[i] = dp[i-1] + dp[i-2]。

18. 背包问题（0/1 背包）

答案：

// 0/1 背包：n 个物品，容量 W，每件物品有重量 weight[i] 和价值 value[i]
function knapsack(weights, values, W) {
  const n = weights.length;
  const dp = new Array(W + 1).fill(0);
  for (let i = 0; i < n; i++) {
    for (let w = W; w >= weights[i]; w--) { // 从后向前防止重复选取
      dp[w] = Math.max(dp[w], dp[w - weights[i]] + values[i]);
    }
  }
  return dp[W];
}
// 一维 DP 优化，时间 O(n*W)，空间 O(W)。

19. 最长公共子序列（LCS，LeetCode 1143）

答案：

function longestCommonSubsequence(text1, text2) {
  const m = text1.length, n = text2.length;
  const dp = Array.from({ length: m + 1 }, () => new Array(n + 1).fill(0));
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      dp[i][j] = text1[i-1] === text2[j-1]
        ? dp[i-1][j-1] + 1
        : Math.max(dp[i-1][j], dp[i][j-1]);
    }
  }
  return dp[m][n];
}
// 二维 DP，时间 O(m*n)，空间 O(m*n)。

21. 快速排序的实现？

答案：

function quickSort(arr, left = 0, right = arr.length - 1) {
  if (left >= right) return arr;
  const pivotIndex = partition(arr, left, right);
  quickSort(arr, left, pivotIndex - 1);
  quickSort(arr, pivotIndex + 1, right);
  return arr;
}

function partition(arr, left, right) {
  const pivot = arr[right];
  let i = left - 1;
  for (let j = left; j < right; j++) {
    if (arr[j] <= pivot) { i++; [arr[i], arr[j]] = [arr[j], arr[i]]; }
  }
  [arr[i + 1], arr[right]] = [arr[right], arr[i + 1]];
  return i + 1;
}
// 平均 O(n log n)，注意随机选 pivot 避免最坏情况。

栈与队列

22. 用两个栈实现队列（LeetCode 232）

答案：

class MyQueue {
  constructor() { this.inStack = []; this.outStack = []; }
  push(x) { this.inStack.push(x); }
  pop() {
    if (!this.outStack.length) {
      while (this.inStack.length) this.outStack.push(this.inStack.pop());
    }
    return this.outStack.pop();
  }
  peek() {
    if (!this.outStack.length) {
      while (this.inStack.length) this.outStack.push(this.inStack.pop());
    }
    return this.outStack[this.outStack.length - 1];
  }
  empty() { return !this.inStack.length && !this.outStack.length; }
}
// 均摊 O(1) 的 push/pop，每个元素最多进出各一次。

23. 前端场景：实现 LRU 缓存（LeetCode 146）

答案：

class LRUCache {
  constructor(capacity) {
    this.capacity = capacity;
    this.cache = new Map(); // Map 保持插入顺序，最旧的在最前面
  }
  get(key) {
    if (!this.cache.has(key)) return -1;
    const val = this.cache.get(key);
    this.cache.delete(key);
    this.cache.set(key, val); // 移到最后（最近使用）
    return val;
  }
  put(key, value) {
    if (this.cache.has(key)) this.cache.delete(key);
    else if (this.cache.size >= this.capacity) {
      this.cache.delete(this.cache.keys().next().value); // 删最旧
    }
    this.cache.set(key, value);
  }
}
// 利用 Map 的有序性，get/put 均 O(1)。前端可用于实现接口缓存层。

双指针

24. 移动零（LeetCode 283）

答案：

function moveZeroes(nums) {
  let insertPos = 0;
  for (const num of nums) {
    if (num !== 0) nums[insertPos++] = num;
  }
  while (insertPos < nums.length) nums[insertPos++] = 0;
}
// 双指针原地操作，时间 O(n)，空间 O(1)。

25. 前端高频：数组去重的多种实现？

答案：

// 1. Set（最简洁，保留第一次出现的顺序）
const unique1 = arr => [...new Set(arr)];

// 2. filter + indexOf（语义清晰）
const unique2 = arr => arr.filter((v, i) => arr.indexOf(v) === i);

// 3. reduce
const unique3 = arr => arr.reduce((acc, v) => acc.includes(v) ? acc : [...acc, v], []);

// 4. 对象去重（NaN 问题需特殊处理）
const unique4 = arr => {
  const seen = {};
  return arr.filter(v => seen.hasOwnProperty(v) ? false : (seen[v] = true));
};
